    <!-- SCRIPTS DE FUNCIONALIDAD -->
    <script>
        // Estado de la aplicación
        let playlist = [];
        let currentIndex = 0;
        let isPlaying = false;
        let isShuffle = false;
        let repeatMode = 0; // 0: No repeat, 1: Repeat All, 2: Repeat One
        let currentFilter = 'songs';

        const audioElement = document.getElementById('audioElement');
        const trackListEl = document.getElementById('trackList');
        const emptyStateEl = document.getElementById('emptyState');
        const fileInput = document.getElementById('localFileInput');

        // Manejar selección de archivos MP3 locales
        fileInput.addEventListener('change', (e) => {
            const files = Array.from(e.target.files);
            addFilesToLibrary(files);
        });

        // Soporte para arrastrar y soltar archivos
        window.addEventListener('dragover', (e) => e.preventDefault());
        window.addEventListener('drop', (e) => {
            e.preventDefault();
            if (e.dataTransfer.files && e.dataTransfer.files.length > 0) {
                const files = Array.from(e.dataTransfer.files).filter(f => f.type.includes('audio') || f.name.endsWith('.mp3'));
                addFilesToLibrary(files);
            }
        });

        function addFilesToLibrary(files) {
            files.forEach((file, index) => {
                // Crear URL de objeto local para reproducir el archivo
                const fileURL = URL.createObjectURL(file);
                
                // Extraer metadatos básicos del nombre del archivo
                let title = file.name.replace(/\.[^/.]+$/, "");
                let artist = "Artista desconocido";
                let album = "Álbum local";

                if (title.includes(" - ")) {
                    const parts = title.split(" - ");
                    artist = parts[0].trim();
                    title = parts[1].trim();
                }

                playlist.push({
                    id: Date.now() + index + Math.random(),
                    title: title,
                    artist: artist,
                    album: album,
                    src: fileURL,
                    duration: "--:--"
                });
            });

            renderTrackList();
            if (playlist.length === files.length && playlist.length > 0) {
                loadTrack(0, false);
            }
        }

        function renderTrackList(filterQuery = '') {
            trackListEl.innerHTML = '';
            
            const filtered = playlist.filter(track => 
                track.title.toLowerCase().includes(filterQuery.toLowerCase()) ||
                track.artist.toLowerCase().includes(filterQuery.toLowerCase()) ||
                track.album.toLowerCase().includes(filterQuery.toLowerCase())
            );

            if (filtered.length === 0) {
                emptyStateEl.classList.remove('hidden');
                emptyStateEl.classList.add('flex');
                return;
            } else {
                emptyStateEl.classList.add('hidden');
                emptyStateEl.classList.remove('flex');
            }

            filtered.forEach((track, idx) => {
                const originalIndex = playlist.findIndex(t => t.id === track.id);
                const isCurrent = (originalIndex === currentIndex);

                const item = document.createElement('div');
                item.className = `flex items-center justify-between p-3 rounded-xl transition group hover:bg-neutral-900 cursor-pointer ${isCurrent ? 'bg-neutral-900/80 border border-neutral-800' : ''}`;
                item.onclick = () => playTrack(originalIndex);

                item.innerHTML = `
                    <div class="flex items-center space-x-4 min-w-0">
                        <div class="w-10 h-10 rounded-lg bg-neutral-800 flex items-center justify-center shrink-0 text-neutral-400 group-hover:bg-neutral-700 transition relative overflow-hidden">
                            ${isCurrent && isPlaying ? 
                                '<div class="flex items-end space-x-0.5 h-4"><span class="w-1 bg-apple-red animate-bounce h-full"></span><span class="w-1 bg-apple-red animate-bounce h-2/3 delay-75"></span><span class="w-1 bg-apple-red animate-bounce h-4/5 delay-150"></span></div>' : 
                                '<i class="fa-solid fa-music text-xs"></i>'}
                        </div>
                        <div class="min-w-0">
                            <h4 class="text-sm font-medium truncate ${isCurrent ? 'text-apple-red font-semibold' : 'text-white'}">${track.title}</h4>
                            <p class="text-xs text-neutral-400 truncate mt-0.5">${track.artist} — <span class="text-neutral-500">${track.album}</span></p>
                        </div>
                    </div>
                    <div class="flex items-center space-x-4 shrink-0 pl-2">
                        <button onclick="event.stopPropagation(); removeTrack(${originalIndex})" class="text-neutral-600 hover:text-red-500 transition opacity-0 group-hover:opacity-100 p-2">
                            <i class="fa-solid fa-trash-can text-xs"></i>
                        </button>
                    </div>
                `;
                trackListEl.appendChild(item);
            });
        }

        function handleSearch(query) {
            renderTrackList(query);
        }

        function filterLibrary(type) {
            currentFilter = type;
            document.querySelectorAll('.nav-btn').forEach(btn => btn.classList.remove('bg-neutral-900/60', 'active-nav'));
            event.currentTarget.classList.add('bg-neutral-900/60', 'active-nav');
            
            const titles = {
                playlists: 'Tus Playlists',
                artists: 'Artistas en Biblioteca',
                albums: 'Álbumes locales',
                songs: 'Todas las Canciones'
            };
            document.getElementById('sectionTitle').innerText = titles[type];
            renderTrackList();
        }

        function loadTrack(index, autoPlay = true) {
            if (playlist.length === 0) return;
            currentIndex = index;
            const track = playlist[currentIndex];

            audioElement.src = track.src;
            audioElement.load();

            // Actualizar interfaz flotante y completa
            document.getElementById('miniTitle').innerText = track.title;
            document.getElementById('miniArtist').innerText = track.artist;
            document.getElementById('fullTitle').innerText = track.title;
            document.getElementById('fullArtist').innerText = track.artist;

            renderTrackList();

            if (autoPlay) {
                audioElement.play().then(() => {
                    isPlaying = true;
                    updatePlayIcons();
                }).catch(e => console.log("Esperando interacción del usuario", e));
            }
        }

        function playTrack(index) {
            loadTrack(index, true);
        }

        function playAll() {
            if (playlist.length > 0) {
                playTrack(0);
            }
        }

        function togglePlayPause() {
            if (playlist.length === 0) return;
            if (isPlaying) {
                audioElement.pause();
                isPlaying = false;
            } else {
                audioElement.play().then(() => {
                    isPlaying = true;
                }).catch(e => console.log(e));
            }
            updatePlayIcons();
            renderTrackList();
        }

        function updatePlayIcons() {
            const iconClass = isPlaying ? "fa-solid fa-pause text-sm" : "fa-solid fa-play text-sm ml-0.5";
            const fullIconClass = isPlaying ? "fa-solid fa-pause text-2xl" : "fa-solid fa-play text-2xl ml-1";

            document.getElementById('playPauseIcon').className = iconClass;
            document.getElementById('fullPlayPauseIcon').className = fullIconClass;
        }

        function nextTrack() {
            if (playlist.length === 0) return;
            if (isShuffle) {
                currentIndex = Math.floor(Math.random() * playlist.length);
            } else {
                currentIndex = (currentIndex + 1) % playlist.length;
            }
            loadTrack(currentIndex, true);
        }

        function prevTrack() {
            if (playlist.length === 0) return;
            currentIndex = (currentIndex - 1 + playlist.length) % playlist.length;
            loadTrack(currentIndex, true);
        }

        // Control de tiempo y eventos de audio
        audioElement.addEventListener('timeupdate', () => {
            if (isNaN(audioElement.duration)) return;
            const progress = (audioElement.currentTime / audioElement.duration) * 100;
            
            document.getElementById('progressBar').value = progress;
            document.getElementById('fullProgressBar').value = progress;

            document.getElementById('currentTime').innerText = formatTime(audioElement.currentTime);
            document.getElementById('fullCurrentTime').innerText = formatTime(audioElement.currentTime);
            document.getElementById('totalDuration').innerText = formatTime(audioElement.duration);
            document.getElementById('fullTotalDuration').innerText = formatTime(audioElement.duration);
        });

        audioElement.addEventListener('ended', () => {
            if (repeatMode === 2) {
                audioElement.currentTime = 0;
                audioElement.play();
            } else {
                nextTrack();
            }
        });

        function seekAudio(val) {
            if (isNaN(audioElement.duration)) return;
            const time = (val / 100) * audioElement.duration;
            audioElement.currentTime = time;
        }

        function setVolume(val) {
            audioElement.volume = val;
        }

        function toggleMute() {
            audioElement.muted = !audioElement.muted;
        }

        function toggleShuffle() {
            isShuffle = !isShuffle;
            const colorClass = isShuffle ? 'text-apple-red' : 'text-neutral-500 hover:text-white';
            document.getElementById('shuffleBtn').className = `${colorClass} transition text-sm`;
            document.getElementById('fullShuffleBtn').className = `${colorClass} transition text-lg`;
        }

        function toggleRepeat() {
            repeatMode = (repeatMode + 1) % 3;
            const repeatBtn = document.getElementById('repeatBtn');
            const fullRepeatBtn = document.getElementById('fullRepeatBtn');
            
            if (repeatMode === 0) {
                repeatBtn.className = 'text-neutral-500 hover:text-white transition text-sm';
                fullRepeatBtn.className = 'text-neutral-500 hover:text-white transition text-lg';
                repeatBtn.innerHTML = '<i class="fa-solid fa-repeat"></i>';
                fullRepeatBtn.innerHTML = '<i class="fa-solid fa-repeat"></i>';
            } else if (repeatMode === 1) {
                repeatBtn.className = 'text-apple-red transition text-sm';
                fullRepeatBtn.className = 'text-apple-red transition text-lg';
                repeatBtn.innerHTML = '<i class="fa-solid fa-repeat"></i>';
                fullRepeatBtn.innerHTML = '<i class="fa-solid fa-repeat"></i>';
            } else {
                repeatBtn.className = 'text-apple-red transition text-sm';
                fullRepeatBtn.className = 'text-apple-red transition text-lg';
                repeatBtn.innerHTML = '<i class="fa-solid fa-repeat-1"></i>';
                fullRepeatBtn.innerHTML = '<i class="fa-solid fa-repeat-1"></i>';
            }
        }

        function removeTrack(index) {
            playlist.splice(index, 1);
            if (currentIndex >= playlist.length && playlist.length > 0) {
                currentIndex = playlist.length - 1;
            }
            renderTrackList();
            if (playlist.length === 0) {
                audioElement.pause();
                audioElement.src = '';
                isPlaying = false;
                updatePlayIcons();
                document.getElementById('miniTitle').innerText = 'Ninguna canción';
                document.getElementById('miniArtist').innerText = 'Selecciona o agrega música';
            }
        }

        function formatTime(seconds) {
            const mins = Math.floor(seconds / 60);
            const secs = Math.floor(seconds % 60);
            return `${mins}:${secs < 10 ? '0' : ''}${secs}`;
        }

        function toggleFullPlayer() {
            const modal = document.getElementById('fullPlayerModal');
            if (modal.classList.contains('hidden-modal')) {
                modal.classList.remove('hidden-modal');
                modal.classList.add('visible-modal');
            } else {
                modal.classList.remove('visible-modal');
                modal.classList.add('hidden-modal');
            }
        }

        function openTrackOptions() {
            alert("Opciones de pista adicionales disponibles próximamente.");
        }
